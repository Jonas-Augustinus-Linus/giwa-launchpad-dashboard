111def build_research(session_allowlist: list[str]) -> list[dict[str, Any]]:
112    if not session_allowlist or len(session_allowlist) != len(set(session_allowlist)):
113        raise ValueError("publication.research_session_allowlist must be a non-empty unique list")
114    allowed = set(session_allowlist)
115    sessions: list[dict[str, Any]] = []
116    discovered: set[str] = set()
117    for state_path in sorted((ROOT / "RESEARCH").glob("*/state.json"), reverse=True):
118        folder = state_path.parent
119        if folder.name not in allowed:
120            continue
121        discovered.add(folder.name)
122        state = load_json(state_path)
123        eval_path = folder / "outputs" / "eval_report.json"
124        evaluation = load_json(eval_path) if eval_path.exists() else {}
125        counts = evaluation.get("counts", {})
126        topic = state.get("topic") or "근거 기반 리서치 세션"
127        title, description = research_copy(folder.name, topic)
128        rel = folder.relative_to(ROOT).as_posix()
129        report_path = f"{rel}/outputs/01_full_report/01_report.ko.md"
130        summary_path = f"{rel}/outputs/00_executive_summary.md"
131        sources_path = f"{rel}/sources/sources.jsonl"
132        ledger_path = f"{rel}/artifacts/claim_ledger.jsonl"
133        sessions.append(
134            {
135                "id": state.get("session_id", folder.name),
136                "title": title,
137                "description": description,
138                "topic": topic,
139                "status": state.get("status", "UNKNOWN"),
140                "updated_at": state.get("updated_at") or state.get("created_at"),
141                "verdict": evaluation.get("verdict", "NOT_EVALUATED"),
142                "source_count": counts.get("sources_total", json_line_count(folder / "sources" / "sources.jsonl")),
143                "claim_count": json_line_count(folder / "artifacts" / "claim_ledger.jsonl"),

155        )
156    missing = allowed - discovered
157    if missing:

159    return sorted(sessions, key=lambda item: item.get("updated_at") or "", reverse=True)

162def parse_gates() -> list[dict[str, str]]:
163    path = ROOT / "docs" / "MAINNET_GATES.md"
164    gates: list[dict[str, str]] = []
165    for line in path.read_text(encoding="utf-8").splitlines():
166        if line.startswith("- "):
167            gates.append({"id": f"gate-{len(gates) + 1}", "state": "blocked", "text": line[2:].strip()})
168        elif gates and line.startswith("  ") and line.strip():
169            gates[-1]["text"] += f" {line.strip()}"

173def validate_observation(item: Any, path: Path) -> dict[str, Any]:
174    if not isinstance(item, dict):
175        raise ValueError(f"unsafe observation {path}: root must be an object")
176    errors: list[str] = []
177    if item.get("chain_id") != GIWA_SEPOLIA_CHAIN_ID:
178        errors.append(f"chain_id must be {GIWA_SEPOLIA_CHAIN_ID}")
179    for key, expected in OBSERVATION_SAFETY.items():
180        if item.get(key) is not expected:
181            errors.append(f"{key} must be {str(expected).lower()}")
182    if not item.get("observed_at"):
183        errors.append("observed_at is required")
184    if errors:
185        raise ValueError(f"unsafe observation {path}: {'; '.join(errors)}")
186    return item

189def latest_observation() -> dict[str, Any] | None:
190    observations: list[dict[str, Any]] = []
191    for path in ROOT.glob(OBSERVATION_GLOB):
192        try:
193            item = load_json(path)
194        except (OSError, json.JSONDecodeError) as exc:
195            raise ValueError(f"invalid observation {path}: {exc}") from exc
196        validate_observation(item, path)
197        item["source_file"] = path.relative_to(ROOT).as_posix()
198        observations.append(item)
199    if not observations:
200        return None
201    return max(observations, key=lambda item: item.get("observed_at", ""))

204def validate_revenue_data(payload: Any, path: Path) -> dict[str, Any]:
205    if not isinstance(payload, dict):
206        raise ValueError(f"unsafe revenue dataset {path}: root must be an object")
207    errors: list[str] = []
208    safety = payload.get("safety") or {}
209    for key, expected in OBSERVATION_SAFETY.items():
210        if safety.get(key) is not expected:
211            errors.append(f"safety.{key} must be {str(expected).lower()}")
212
213    session = path.parents[1]
214    state = load_json(session / "state.json")
215    evaluation = load_json(session / "outputs" / "eval_report.json")
216    if state.get("status") not in {"COMPLETE", "COMPLETED"}:
217        errors.append("owning research session must be complete")
218    if not state.get("verification", {}).get("passed"):
219        errors.append("owning research verification must pass")
220    if evaluation.get("verdict") != "PASS":
221        errors.append("owning research evaluation must pass")
222
223    source_rows: list[dict[str, Any]] = []
224    sources_path = session / "sources" / "sources.jsonl"
225    try:
226        with sources_path.open(encoding="utf-8") as handle:
227            source_rows = [json.loads(line) for line in handle if line.strip()]
228    except (OSError, json.JSONDecodeError) as exc:
229        errors.append(f"source registry is invalid: {exc}")
230    known_sources = {row.get("id") for row in source_rows}
231    for collection_name in ("models", "case_studies"):
232        collection = payload.get(collection_name)
233        if not isinstance(collection, list) or not collection:
234            errors.append(f"{collection_name} must be a non-empty list")
235            continue
236        for item in collection:
237            item_id = item.get("id", "unknown") if isinstance(item, dict) else "unknown"
238            source_ids = item.get("source_ids", []) if isinstance(item, dict) else []
239            unknown = set(source_ids) - known_sources
240            if not source_ids:
241                errors.append(f"{collection_name}.{item_id} requires source_ids")
242            elif unknown:
243                errors.append(f"{collection_name}.{item_id} has unknown source_ids {sorted(unknown)}")
244    if errors:
245        raise ValueError(f"unsafe revenue dataset {path}: {'; '.join(errors)}")
246    return payload

287def catalog_payload() -> dict[str, Any]:
288    manifest = load_json(MANIFEST_PATH)
289    ecosystem = load_json(ECOSYSTEM_PATH)
290    project = manifest["project"]
291    branch = os.environ.get("DASHBOARD_REPO_REF") or project["default_branch"]
292    project = dict(project)
293    project.pop("repo_url", None)
294    project["content_ref"] = branch
295    overrides = manifest.get("ecosystem_overrides", {})
296
297    ecosystem_entries: list[dict[str, Any]] = []
298    for raw_entry in ecosystem["entries"]:
299        entry = dict(raw_entry)
300        override = overrides.get(entry["id"], {})
301        entry["code_state"] = override.get("code_state", "unknown")
302        if entry.get("contracts"):
303            entry["address_state"] = override.get("address_state", "catalogued_unverified")
304        else:
305            entry["address_state"] = "not_applicable"
306        entry["links"] = override.get("links", [])
307        entry["source_count"] = len(entry.get("source_ids", []))
308        entry["contract_count"] = len(entry.get("contracts", []))
309        ecosystem_entries.append(entry)
310
311    publication = manifest.get("publication") or {}
312    research = build_research(publication.get("research_session_allowlist", []))
313    build_tracks = [dict(item) for item in manifest.get("build_tracks", [])]
314    add_repo_links(build_tracks)
315    document_groups = [dict(item) for item in manifest.get("document_groups", [])]
316    add_repo_links(document_groups)
317
318    official_count = sum(item["officiality"] == "official" for item in ecosystem_entries)
319    provider_count = sum(item["officiality"] == "provider_reported" for item in ecosystem_entries)
320    third_party_count = sum(item["officiality"] == "third_party" for item in ecosystem_entries)
321    total_sources = sum(item["source_count"] for item in research)
322    total_verified = sum(item["verified_count"] for item in research)
323    total_unresolved = sum(item["unresolved_count"] for item in research)
324    total_refuted = sum(item["refuted_count"] for item in research)
325    contract_count = sum(item["contract_count"] for item in ecosystem_entries)
326    code_ready_count = sum(item["status"] == "implemented" for item in build_tracks)
327
328    fingerprint_inputs = [Path(__file__), MANIFEST_PATH, ECOSYSTEM_PATH, LAUNCH_PREP_PATH]
329    fingerprint_inputs.extend(sorted((ROOT / "RESEARCH").glob("*/state.json")))
330    fingerprint_inputs.extend(sorted((ROOT / "RESEARCH").glob("*/outputs/eval_report.json")))
331    fingerprint_inputs.extend(sorted((ROOT / "RESEARCH").glob("*/sources/sources.jsonl")))
332    fingerprint_inputs.extend(sorted((ROOT / "RESEARCH").glob("*/artifacts/claim_ledger.jsonl")))
333    fingerprint_inputs.extend(sorted((ROOT / "RESEARCH").glob("*/outputs/**/*.md")))
334    fingerprint_inputs.extend(sorted(ROOT.glob(REVENUE_GLOB)))
335    fingerprint = hashlib.sha256(branch.encode("utf-8"))
336    for path in fingerprint_inputs:
337        if path.is_file():
338            fingerprint.update(path.relative_to(ROOT).as_posix().encode("utf-8"))
339            fingerprint.update(path.read_bytes())
340    source_fingerprint = fingerprint.hexdigest()[:12]
341
342    return {
343        "schema_version": 1,
344        "catalog_revision": source_fingerprint,
345        "as_of": manifest["as_of"],
346        "project": project,
347        "publication": publication,
348        "safety": manifest["safety"],
349        "launch_prep": launch_prep_data(),
350        "metrics": {
351            "ecosystem_entries": len(ecosystem_entries),
352            "official_entries": official_count,
353            "provider_entries": provider_count,
354            "third_party_entries": third_party_count,
355            "catalogued_addresses": contract_count,
356            "research_sessions": len(research),
357            "registered_sources": total_sources,
358            "verified_claims": total_verified,
359            "unresolved_claims": total_unresolved,
360            "refuted_claims": total_refuted,
361            "implemented_tracks": code_ready_count,
362            "total_tracks": len(build_tracks),
363        },
364        "network_observation": latest_observation(),
365        "official_resources": manifest.get("official_resources", []),
366        "ecosystem": ecosystem_entries,
367        "infrastructure": manifest.get("infrastructure", {}),
368        "pool_blueprint": manifest.get("pool_blueprint", {}),
369        "revenue": latest_revenue_data(),
370        "research": research,
371        "build_tracks": build_tracks,
372        "document_groups": document_groups,
373        "mainnet_gates": parse_gates(),
374    }

381def main() -> int:
382    parser = argparse.ArgumentParser(description=__doc__)
383    parser.add_argument("--check", action="store_true", help="fail if the checked-in catalog is stale")
384    args = parser.parse_args()
385    expected = serialized_catalog()
386    if args.check:
387        if not OUTPUT_PATH.exists() or OUTPUT_PATH.read_text(encoding="utf-8") != expected:
388            print(f"dashboard catalog is stale: run {Path(__file__).relative_to(ROOT)}")
389            return 1
390        print(f"dashboard catalog is current ({catalog_payload()['catalog_revision']})")
391        return 0
392    OUTPUT_PATH.parent.mkdir(parents=True, exist_ok=True)
393    OUTPUT_PATH.write_text(expected, encoding="utf-8")
394    print(f"wrote {OUTPUT_PATH.relative_to(ROOT)}")
395    return 0
