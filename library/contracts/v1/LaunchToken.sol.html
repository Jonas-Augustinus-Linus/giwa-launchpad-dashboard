9///      upgrade, pause or transfer-tax path. launchpad is stored so runtime code is identical.

11    uint256 public constant SUPPLY = 1_000_000_000e18;
12    address public launchpad;
13
14    error InvalidMetadata();
15
16    constructor(string memory name_, string memory symbol_, address curve_) ERC20(name_, symbol_) {
17        if (bytes(name_).length == 0 || bytes(name_).length > 64 ||
18            bytes(symbol_).length == 0 || bytes(symbol_).length > 16) revert InvalidMetadata();
19        launchpad = curve_;
20        _mint(curve_, SUPPLY); // OpenZeppelin rejects the zero address.
21    }
22}
