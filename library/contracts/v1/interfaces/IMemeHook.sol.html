4/// @title IMemeHook — the Uniswap v4 hook attached to every graduated pool on Robinhood Chain (launchpad v1)
5/// @notice Singleton hook, NO owner, NO setters, NO rescue. Per-pool terms are written exactly once by the
6///         graduation target (`register`) BEFORE `PoolManager.initialize`; `beforeInitialize` reverts unless
7///         `sender == TARGET` and the key is registered and not yet initialised, so nobody can pre-create our
8///         pool key at another price and exactly ONE hooked pool exists per token forever.
9/// @dev    Permissions (low 14 bits of the hook address) = 0x22CC:
10///           beforeInitialize (1<<13) | beforeRemoveLiquidity (1<<9) | beforeSwap (1<<7) | afterSwap (1<<6)
11///           | beforeSwapReturnDelta (1<<3) | afterSwapReturnDelta (1<<2).
12///           Drop beforeRemoveLiquidity (defence-in-depth only) => 0x20CC. Flag values from
13///           https://github.com/Uniswap/v4-core/blob/main/src/libraries/Hooks.sol ; deployment rule from
14///           https://developers.uniswap.org/docs/protocols/v4/concepts/hooks (address encodes permissions).
15///         Fee (always in the QUOTE, never in the memecoin — Clanker pattern, not Pons' afterSwap-only):
16///           quote is the *specified* currency  (exact-in buy, exact-out sell)  -> beforeSwap  + beforeSwapReturnDelta
17///           quote is the *unspecified* currency (exact-out buy, exact-in sell) -> afterSwap   + afterSwapReturnDelta
18///           fee = bps * G where G = gross quote on the paying side; exact-out cases use fee = net * p / (1 - p),
19///           rounded up, so exact-in and exact-out quote identically (custom-accounting guide:
20///           https://developers.uniswap.org/docs/protocols/v4/guides/custom-accounting).
21///           bps = BASE_FEE_BPS (100) + snipe(t) on buys, where t = block.timestamp - initTimestamp, whole seconds,
22///           SnipeSchedule table [9900, 9900, 2475, 309, 38] then 0; total capped at MAX_TOTAL_BPS (9900).
23///           Sells never pay more than 100 bps. No creator tax post-graduation (curve-only). No exempt addresses.
24///         Settlement per swap: poolManager.take(quote, hook, fee) then
25///           ESCROW.credit{value: base}(token, base, 0, snipe)  [shared 80/20 + snipe 100 % protocol].
26///           If the escrow call reverts (must be impossible for a registered + bound token — tested), the fee is
27///           kept in `stranded[token]` and `FeeStranded` is emitted; anyone can `flush(token)` later. A swap is
28///           therefore NEVER reverted by the fee path.
29///         Pool LP fee is static 0; third-party liquidity is allowed (earns nothing); `donate` is allowed (value
30///           accrues to in-range liquidity, i.e. the uncollectable locked position).
31///         beforeRemoveLiquidity: reverts iff sender == POSITION_MANAGER && params.salt == bytes32(lockedId) &&
32///           liquidityDelta < 0 for either locked id read from LOCKER.lockInfo(token). PositionManager passes
33///           salt = bytes32(tokenId) (https://github.com/Uniswap/v4-periphery/blob/main/src/PositionManager.sol).
34///         The implementation also implements v4-core `IHooks` (beforeInitialize / beforeRemoveLiquidity /
35///           beforeSwap / afterSwap; all other callbacks revert HookNotImplemented). Those callbacks use v4 types
36///           and are not repeated here so this file compiles in isolation.
37///         Deployment: CreateX `deployCreate2` with a permissioned salt whose 11 entropy bytes are mined so that
38///           keccak256(0xFF ++ CreateX ++ guardedSalt ++ keccak256(initCode)) has low bits 0x22CC. Robinhood only.
39interface IMemeHook {
40    // ------------------------------------------------------------------ types
41    /// @notice ABI-identical to v4-core `PoolKey` (address,address,uint24,int24,address). For our pools:
42    ///         currency0 = address(0) (native ETH), currency1 = token, fee = 0, tickSpacing = 200, hooks = this.
43    struct PoolKey {
44        address currency0;
45        address currency1;
46        uint24  fee;
47        int24   tickSpacing;
48        address hooks;
49    }
50
51    struct PoolTerms {
52        address token;            // the launched memecoin
53        bool    tokenIsCurrency0; // always false for ETH pairs (address(0) sorts first); kept for ERC-20 quotes (v1.1)
54        bool    initialized;      // set in beforeInitialize
55        uint40  initTimestamp;    // block.timestamp at beforeInitialize; post-graduation snipe anchor
56    }
57
58    // ------------------------------------------------------------------ events
59    event PoolRegistered(address indexed token, bytes32 indexed poolId, address quote, bool tokenIsCurrency0);
60    event PoolInitialized(bytes32 indexed poolId, address indexed token, uint40 initTimestamp, uint160 sqrtPriceX96);
61    /// @notice Source of truth for creator revenue indexers. grossQuote * bps / 10_000 == baseFee + snipeFee.
62    event FeeTaken(
63        bytes32 indexed poolId, address indexed token, bool isBuy, bool exactIn,
64        uint256 grossQuote, uint256 baseFee, uint256 snipeFee
65    );
66    /// @notice The escrow refused a credit; the fee is held in the hook until flush(token) succeeds.
67    event FeeStranded(address indexed token, uint256 amount, bytes reason);
68    event FeeFlushed(address indexed token, uint256 amount);
69
70    // ------------------------------------------------------------------ errors
71    error OnlyTarget();
72    error OnlyPoolManager();
73    error AlreadyRegistered(address token);
74    error NotRegistered(bytes32 poolId);
75    error AlreadyInitialized(bytes32 poolId);
76    error BadPoolKey();                 // fee != 0, tickSpacing != TICK_SPACING, hooks != this, or currency mismatch
77    error QuoteMismatch(address expected, address got); // escrow.quoteOf(token) != the key's quote currency
78    error VenueMismatch();              // escrow.venueOf(token) != address(this)
79    error LockedPosition(uint256 tokenId);
80    error NothingStranded(address token);
81    error InexactQuoteReceipt();        // ERC-20 quote delivered less than `fee` (fee-on-transfer) — v1 quote is native only
82
83    // ------------------------------------------------------------------ target-only
84    /// @notice Registers the canonical pool for `token`. Called by TARGET in the graduation transaction, BEFORE
85    ///         `PoolManager.initialize(key, sqrtPriceX96)`. Verifies key shape, ESCROW.quoteOf(token) and
86    ///         ESCROW.venueOf(token) == address(this).
87    function register(PoolKey calldata key, address token) external;
88
89    // ------------------------------------------------------------------ permissionless
90    /// @notice Retries crediting a stranded fee to the escrow.
91    function flush(address token) external;
92
93    // ------------------------------------------------------------------ views
94    function terms(bytes32 poolId) external view returns (PoolTerms memory);
95    /// @notice The one hooked pool for `token` (bytes32(0) if none).
96    function poolOf(address token) external view returns (bytes32 poolId);
97    /// @notice Total fee bps a swap would pay right now (100 + snipe on buys; 100 on sells).
98    function feeBpsNow(bytes32 poolId, bool isBuy) external view returns (uint256 bps);
99    /// @notice Fee in quote for a hypothetical swap: exactIn => amount is the quote in/out; exactOut => amount is the
100    ///         quote the pool would pay/take, fee = amount * bps / (10_000 - bps) rounded up.
101    function quoteFee(bytes32 poolId, bool isBuy, bool exactIn, uint256 amount) external view returns (uint256 fee);
102    function stranded(address token) external view returns (uint256);
103
104    // immutables / constants
105    function POOL_MANAGER() external view returns (address);      // 0x8366a39CC670B4001A1121B8F6A443A643e40951 (4663)
106    function POSITION_MANAGER() external view returns (address);  // 0x58daec3116aae6D93017bAAea7749052E8a04fA7 (4663)
107    function TARGET() external view returns (address);            // UniswapV4Target (CREATE3)
108    function ESCROW() external view returns (address);            // FeeEscrow (CREATE3)
109    function LOCKER() external view returns (address);            // PermanentLocker (CREATE3)
110    function TICK_SPACING() external view returns (int24);        // 200
111    function BASE_FEE_BPS() external view returns (uint16);       // 100
112    function MAX_TOTAL_BPS() external view returns (uint16);      // 9_900
113    function SNIPE_WINDOW_SECONDS() external view returns (uint8);// 5
114    function HOOK_FLAGS() external view returns (uint16);         // 0x22CC
115}
