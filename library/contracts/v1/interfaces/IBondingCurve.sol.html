4/// @title IBondingCurve — the per-chain singleton curve (launchpad v1)
5/// @notice ONE contract per chain holds the curve state of every token in a mapping (no clones, no per-launch
6///         contracts). Quote asset = native ETH on both chains. Fees are NOT held here: every trade credits
7///         FeeEscrow in the same call (`credit{value}`), so the escrow is the only ledger.
8/// @dev    Curve: (V + R) * (X0 - S) = V * X0   (virtual quote only; V and Q per immutable tier)
9///           R = real quote held for the token, S = tokens sold, X0 = 1e27.
10///           Graduation trigger: S == X_SELL where X_SELL = X0 * Q / (V + Q) (identical for all tiers because
11///           Q = 2.5 V in every tier) -> 714,285,714.285714 tokens (71.4286 %). X_RES = X0 - X_SELL.
12///         Rounding: every amount paid OUT by the curve is floored; the crossing buy's required input is
13///           ceiled (+1 wei) and grossed up with ceil; fees floor on the gross leg. Hence k' >= k always.
14///         Snipe tax (buys only): SnipeSchedule.snipeBps(block.timestamp - launchedAt), whole seconds,
15///           table [9900, 9900, 2475, 309, 38] then 0 (t >= 5). Capped so total bps <= MAX_TOTAL_BPS (9900).
16///           100 % of snipe proceeds go to the protocol (escrow `protocolOnly` leg). No exempt addresses;
17///           the only exempt buy is the dev buy executed inside `open()` (call-path exemption).
18///         Reentrancy: OZ ReentrancyGuardTransient on buy/sell/graduate. The atomic graduation attempt inside
19///           the crossing buy uses the INTERNAL unguarded path (`_graduate`) so it does not re-enter the guard;
20///           the permissionless `graduate()` retry takes the guard then calls the same internal path.
21///         Status machine per token: None -> Trading -> Closed -> Graduated. There is NO wind-down / rescue state.
22interface IBondingCurve {
23    // ------------------------------------------------------------------ types
24    enum Status { None, Trading, Closed, Graduated }
25
26    /// @notice Immutable tier table entry (all tiers keep graduationQuote == 2.5 * virtualQuote).
27    struct Tier {
28        uint128 virtualQuote;     // V, wei  (tier0 0.6e18, tier1 1.68e18, tier2 2.8e18)
29        uint128 graduationQuote;  // Q, wei  (tier0 1.5e18, tier1 4.2e18, tier2 7.0e18)
30    }
31
32    /// @notice Per-token state (2 storage slots).
33    struct CurveState {
34        uint128 realQuote;        // R, wei; 0 once Graduated (moved to the pool)
35        uint128 tokensSold;       // S, <= X_SELL
36        uint64  launchedAt;       // block.timestamp at open(); snipe-tax anchor
37        uint64  closedAt;         // block.timestamp of the crossing buy (0 before)
38        uint8   tier;
39        uint16  creatorTaxBps;    // 0..200, curve-only
40        Status  status;
41    }
42
43    struct BuyQuote {
44        uint256 tokensOut;
45        uint256 quoteSpent;       // gross quote actually consumed (== quoteIn unless the buy crosses)
46        uint256 quoteRefund;      // quoteIn - quoteSpent (crossing buy only)
47        uint256 fee;              // 1 % of quoteSpent -> escrow shared (80/20)
48        uint256 creatorTax;       // creatorTaxBps of quoteSpent -> escrow creatorOnly
49        uint256 snipeTax;         // snipeBps of quoteSpent -> escrow protocolOnly
50        bool    crosses;          // true if this buy exhausts X_SELL (graduation)
51    }
52
53    struct SellQuote {
54        uint256 quoteOut;         // net to the seller
55        uint256 fee;              // 1 % of gross quote -> escrow shared
56        uint256 creatorTax;       // creatorTaxBps of gross -> escrow creatorOnly
57    }
58
59    // ------------------------------------------------------------------ events
60    event CurveOpened(address indexed token, uint8 tier, uint16 creatorTaxBps, uint64 launchedAt);
61    /// @notice The single disclosed, snipe-exempt dev buy (executed inside open()).
62    event DevBuy(address indexed token, address indexed launcher, uint256 quoteIn, uint256 tokensOut);
63    event Buy(
64        address indexed token, address indexed sender, address indexed recipient,
65        uint256 quoteIn, uint256 quoteSpent, uint256 tokensOut, uint256 fee, uint256 creatorTax, uint256 snipeTax
66    );
67    event Sell(
68        address indexed token, address indexed sender, address indexed recipient,
69        uint256 tokensIn, uint256 quoteOut, uint256 fee, uint256 creatorTax
70    );
71    event Refund(address indexed token, address indexed to, uint256 amount);
72    /// @notice The crossing buy landed: S == X_SELL, buys and sells are closed, graduation follows.
73    event CurveClosed(address indexed token, uint256 realQuote, uint256 tokensSold);
74    /// @notice The atomic graduation attempt reverted; anyone may call graduate(token).
75    event GraduationDeferred(address indexed token, bytes reason, uint256 gasLeft);
76    event Graduated(
77        address indexed token, bytes32 indexed poolId, address venue,
78        uint256 positionIdFull, uint256 positionIdOverhang,
79        uint256 quoteSeeded, uint256 tokensSeeded, uint256 tokensOverhang, uint160 sqrtPriceX96
80    );
81
82    // ------------------------------------------------------------------ errors
83    error NotFactory();
84    error UnknownToken(address token);
85    error NotTrading(Status current);
86    error NotClosed(Status current);
87    error ZeroAmount();
88    error ZeroAddress();
89    error Expired(uint256 deadline);
90    error SlippageExceeded(uint256 got, uint256 min);
91    error DevBuyExceedsCap(uint256 tokensOut, uint256 cap);
92    error CreatorTaxTooHigh(uint16 bps, uint16 max);
93    error InvalidTier(uint8 tier);
94    error AlreadyOpen(address token);
95    error QuoteTransferFailed();
96    error Reentrancy();
97
98    // ------------------------------------------------------------------ factory-only
99    /// @notice Opens the curve for `token` (whose whole supply must already sit in this contract), records
100    ///         `launchedAt`, approves TARGET for X_RES (pulled at graduation), and runs the dev buy with
101    ///         `msg.value` (may be 0) as a snipe-exempt buy to `launcher`. Reverts if tokensOut > MAX_DEV_BUY_TOKENS.
102    function open(address token, uint8 tier, uint16 creatorTaxBps, address launcher, uint256 minDevTokensOut)
103        external payable returns (uint256 devTokensOut);
104
105    // ------------------------------------------------------------------ trading (permissionless)
106    /// @notice Exact-input buy with a price-bound slippage check: spent * minTokensOut <= quoteIn * tokensOut,
107    ///         so a partial (crossing) fill honours the caller's price instead of reverting. Refund -> msg.sender.
108    function buy(address token, uint256 minTokensOut, address recipient, uint256 deadline)
109        external payable returns (uint256 tokensOut, uint256 quoteSpent, uint256 quoteRefunded);
110
111    /// @notice Exact-input sell (Trading only). Never snipe-taxed. 1 % fee + creator tax on the gross quote.
112    function sell(address token, uint256 tokensIn, uint256 minQuoteOut, address recipient, uint256 deadline)
113        external returns (uint256 quoteOut);
114
115    /// @notice Permissionless retry of a deferred graduation (status Closed). Effects: status = Graduated,
116    ///         realQuote = 0, then TARGET.graduate{value: R, gas: GRADUATE_GAS}(token, X_RES, R, V);
117    ///         the target PULLS X_RES via transferFrom inside its own frame. A revert leaves the token Closed.
118    function graduate(address token) external;
119
120    // ------------------------------------------------------------------ views
121    function getState(address token) external view returns (CurveState memory);
122    function tierParams(uint8 tier) external view returns (Tier memory);
123    function tierCount() external view returns (uint8);
124    function quoteBuy(address token, uint256 quoteIn) external view returns (BuyQuote memory);
125    function quoteSell(address token, uint256 tokensIn) external view returns (SellQuote memory);
126    /// @notice Snipe bps that a buy sent NOW would pay (already capped against fee + tax).
127    function currentSnipeBps(address token) external view returns (uint256);
128    /// @notice Marginal price (V + R) / (X0 - S) as quote-per-token, 1e18-scaled.
129    function price(address token) external view returns (uint256 quotePerTokenWad);
130    function sellableTokens(address token) external view returns (uint256);
131    /// @notice S * 10_000 / X_SELL.
132    function graduationProgressBps(address token) external view returns (uint256);
133
134    // immutables / constants
135    function X0() external view returns (uint256);                 // 1e27
136    function X_SELL() external view returns (uint256);             // 714_285_714_285714285714285714 (= X0*5/7)
137    function X_RES() external view returns (uint256);              // X0 - X_SELL
138    function FEE_BPS() external view returns (uint16);             // 100
139    function MAX_CREATOR_TAX_BPS() external view returns (uint16); // 200
140    function MAX_TOTAL_BPS() external view returns (uint16);       // 9_900 (buyer always keeps >= 1 %)
141    function MAX_DEV_BUY_TOKENS() external view returns (uint256); // 50_000_000e18 (5 % of supply)
142    function SNIPE_WINDOW_SECONDS() external view returns (uint8); // 5
143    function GRADUATE_GAS() external view returns (uint256);       // explicit stipend for the atomic attempt (>= 2x measured)
144    function FACTORY() external view returns (address);
145    function TARGET() external view returns (address);
146    function ESCROW() external view returns (address);
147}
