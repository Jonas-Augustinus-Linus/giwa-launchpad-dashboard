17def validate_inventory(document: Any) -> list[str]:
18    errors: list[str] = []
19    if not isinstance(document, dict):
20        return ["inventory root must be an object"]
21    if document.get("schema_version") != 1:
22        errors.append("schema_version must be 1")
23    entries = document.get("entries")
24    if not isinstance(entries, list):
25        return errors + ["entries must be an array"]
26
27    seen: set[str] = set()
28    for index, entry in enumerate(entries):
29        label = f"entries[{index}]"
30        if not isinstance(entry, dict):
31            errors.append(f"{label} must be an object")
32            continue
33        entry_id = entry.get("id")
34        if not isinstance(entry_id, str) or not entry_id:
35            errors.append(f"{label}.id must be a non-empty string")
36        elif entry_id in seen:
37            errors.append(f"{label}.id is duplicated: {entry_id}")
38        else:
39            seen.add(entry_id)
40        if entry.get("officiality") not in OFFICIALITY:
41            errors.append(f"{label}.officiality is invalid")
42        source_ids = entry.get("source_ids")
43        if not isinstance(source_ids, list) or not source_ids or not all(
44            isinstance(item, str) and item.startswith("src_") for item in source_ids
45        ):
46            errors.append(f"{label}.source_ids must contain registered source IDs")
47        for field, expected in REQUIRED_SAFETY_FLAGS.items():
48            if entry.get(field) is not expected:
49                errors.append(f"{label}.{field} must be {expected!r}")
50    return errors

53def load_inventory(path: str | Path) -> dict[str, Any]:
54    with Path(path).open(encoding="utf-8") as handle:
55        document = json.load(handle)
56    errors = validate_inventory(document)
57    if errors:
58        raise ValueError("invalid ecosystem inventory: " + "; ".join(errors))
59    return document
