21def _hex_int(value: Any, field: str) -> int:
22    if not isinstance(value, str) or not value.startswith("0x"):
23        raise RpcResponseError(f"{field} is not a hexadecimal quantity")
24    try:
25        return int(value, 16)
26    except ValueError as exc:
27        raise RpcResponseError(f"{field} is not a valid hexadecimal quantity") from exc

30def collect_probe(client: RpcLike, addresses: list[str] | None = None) -> dict[str, Any]:
31    chain_id = _hex_int(client.request("eth_chainId"), "chain_id")
32    if chain_id != GIWA_SEPOLIA_CHAIN_ID:
33        raise RpcSafetyError(
34            f"refusing non-GIWA-Sepolia chain: expected {GIWA_SEPOLIA_CHAIN_ID}, got {chain_id}"
35        )
36
37    block_number = _hex_int(client.request("eth_blockNumber"), "block_number")
38    block = client.request("eth_getBlockByNumber", [hex(block_number), False])
39    if not isinstance(block, dict):
40        raise RpcResponseError("latest block result is not an object")
41
42    checked_addresses: dict[str, dict[str, Any]] = {}
43    for address in addresses or []:
44        if not ADDRESS_RE.fullmatch(address):
45            raise RpcSafetyError(f"invalid EVM address: {address}")
46        code = client.request("eth_getCode", [address, "latest"])
47        if not isinstance(code, str) or not code.startswith("0x"):
48            raise RpcResponseError(f"invalid bytecode response for {address}")
49        checked_addresses[address] = {
50            "has_code": code not in {"0x", "0x0"},
51            "bytecode_bytes": max(0, (len(code) - 2) // 2),
52        }
53
54    timestamp = _hex_int(block.get("timestamp"), "block.timestamp")
55    return {
56        "observed_at": datetime.now(timezone.utc).isoformat(),
57        "network": "GIWA Sepolia",
58        "chain_id": chain_id,
59        "client_version": client.request("web3_clientVersion"),
60        "syncing": client.request("eth_syncing"),
61        "block_number": block_number,
62        "block_timestamp": datetime.fromtimestamp(timestamp, timezone.utc).isoformat(),
63        "transaction_count": len(block.get("transactions", [])),
64        "addresses": checked_addresses,
65        "research_only": True,
66        "strategy_consumption_allowed": False,
67        "order_input": False,
68        "interpretation_warning": "Testnet activity is not evidence of users, adoption, or value.",
69    }
