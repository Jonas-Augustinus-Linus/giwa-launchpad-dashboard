22def validate_rpc_url(url: str) -> str:
23    """Accept HTTPS, or HTTP only for an explicit loopback development endpoint."""
24
25    parsed = urlsplit(url)
26    loopback = parsed.hostname in {"localhost", "127.0.0.1", "::1"}
27    if parsed.scheme != "https" and not (parsed.scheme == "http" and loopback):
28        raise RpcSafetyError("RPC must use HTTPS (HTTP is allowed only on loopback)")
29    if not parsed.hostname or parsed.username or parsed.password:
30        raise RpcSafetyError("RPC URL must have a host and must not contain credentials")
31    if parsed.query or parsed.fragment:
32        raise RpcSafetyError("RPC URL must not contain query parameters or fragments")
33    return url

36@dataclass
37class JsonRpcClient:
38    endpoint: str
39    timeout: float = 10.0
40    opener: Callable[..., Any] = urlopen

42    def __post_init__(self) -> None:
43        self.endpoint = validate_rpc_url(self.endpoint)
44        if self.timeout <= 0 or self.timeout > 30:
45            raise RpcSafetyError("RPC timeout must be in the range (0, 30] seconds")
46
47    def request(self, method: str, params: list[Any] | None = None) -> Any:
48        if method not in READ_ONLY_RPC_METHODS:
49            raise RpcSafetyError(f"RPC method is not read-only allowlisted: {method}")
50
51        payload = json.dumps(
52            {"jsonrpc": "2.0", "id": 1, "method": method, "params": params or []},
53            separators=(",", ":"),
54        ).encode("utf-8")
55        request = Request(
56            self.endpoint,
57            data=payload,
58            headers={"Content-Type": "application/json", "User-Agent": "giwa-lab/0.1 read-only"},
59            method="POST",
60        )
61        try:
62            with self.opener(request, timeout=self.timeout) as response:
63                body = response.read()
64            decoded = json.loads(body)
65        except (OSError, UnicodeDecodeError, json.JSONDecodeError) as exc:
66            raise RpcResponseError("RPC transport or JSON response failed") from exc
67
68        if not isinstance(decoded, dict) or decoded.get("jsonrpc") != "2.0":
69            raise RpcResponseError("RPC response is not a JSON-RPC 2.0 object")
70        if decoded.get("error") is not None:
71            raise RpcResponseError(f"RPC returned an error: {decoded['error']!r}")
72        if "result" not in decoded:
73            raise RpcResponseError("RPC response has no result")
74        return decoded["result"]
