27const normalizeSafeUrl = (value = "") => {
28  const raw = String(value).trim();
29  const localPath = raw.split(/[?#]/, 1)[0];
30  if (
31    /^\.\/library\/[A-Za-z0-9._/-]+\.html(?:#[A-Za-z0-9._:-]+)?$/.test(raw)
32    && !localPath.split("/").includes("..")
33  ) {
34    return raw;
35  }
36  try {
37    const parsed = new URL(raw);
38    if (["http:", "https:"].includes(parsed.protocol) && !parsed.username && !parsed.password) {
39      return parsed.href;
40    }
41  } catch {
42    // Invalid and non-allowlisted URLs are rendered as inert links.
43  }
44  return "#";
45};

96function codePresentation(codeState) {
97  if (["official_source", "third_party_source", "source_available", "example_available"].includes(codeState)) {
98    return ["code-source", "source available"];
99  }
100  if (["docs_only", "documentation", "addresses_and_docs"].includes(codeState)) {
101    return ["code-docs", "docs / reference"];
102  }
103  if (["external_service", "provider_integration", "provider_reported", "managed_service", "not_applicable"].includes(codeState)) {
104    return ["code-service", "service / external"];
105  }
106  if (["source_rejected", "verified_contracts_no_repo"].includes(codeState)) {
107    return ["code-rejected", codeState === "source_rejected" ? "source rejected" : "no repo found"];
108  }
109  return ["code-docs", String(codeState || "unknown").replaceAll("_", " ")];
110}

287function calculatorValues() {
288  const volume = Math.max(0, Number($("#calc-volume").value) || 0);
289  const feeBps = Math.max(0, Number($("#calc-fee").value) || 0);
290  const share = Math.min(100, Math.max(0, Number($("#calc-share").value) || 0));
291  const cost = Math.max(0, Number($("#calc-cost").value) || 0);
292  const gross = volume * feeBps / 10000;
293  const recipient = gross * share / 100;
294  const net = recipient - cost;
295  const takeRate = feeBps / 10000 * share / 100;
296  const breakEven = takeRate > 0 ? cost / takeRate : 0;
297  return { gross, recipient, net, breakEven };
298}

427function bindInteractions() {
428  $("#copy-probe").addEventListener("click", () => copyText(PROBE_COMMAND, "읽기 전용 프로브 명령을 복사했습니다."));
429  $("#copy-checks").addEventListener("click", () => copyText(CHECK_COMMANDS, "전체 검증 명령을 복사했습니다."));
430  ["#calc-volume", "#calc-fee", "#calc-share", "#calc-cost"].forEach((selector) => {
431    $(selector).addEventListener("input", renderCalculator);
432  });
433
434  $("#ecosystem-search").addEventListener("input", (event) => {
435    state.query = event.target.value;
436    renderEcosystem();
437  });
438
439  $$(".filter-chip").forEach((button) => button.addEventListener("click", () => {
440    state.filter = button.dataset.filter;
441    $$(".filter-chip").forEach((item) => item.classList.toggle("is-active", item === button));
442    renderEcosystem();
443  }));
444
445  $("#ecosystem-rows").addEventListener("click", (event) => {
446    const row = event.target.closest("[data-entry-id]");
447    if (row) openDrawer(row.dataset.entryId, row);
448  });
449  $("#ecosystem-rows").addEventListener("keydown", (event) => {
450    const row = event.target.closest("[data-entry-id]");
451    if (row && ["Enter", " "].includes(event.key)) {
452      event.preventDefault();
453      openDrawer(row.dataset.entryId, row);
454    }
455  });
456
457  $$('[data-close-drawer]').forEach((button) => button.addEventListener("click", closeDrawer));
458  document.addEventListener("keydown", (event) => {
459    if (event.key === "Escape") closeDrawer();
460    if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === "k") {
461      event.preventDefault();
462      $("#ecosystem-search").focus();
463      $("#ecosystem").scrollIntoView({ behavior: "smooth", block: "start" });
464    }
465  });
466
467  const sections = $$("main section[id]");
468  const navLinks = $$(".nav-link");
469  const observer = new IntersectionObserver((entries) => {
470    const visible = entries.filter((entry) => entry.isIntersecting).sort((a, b) => b.intersectionRatio - a.intersectionRatio)[0];
471    if (!visible) return;
472    navLinks.forEach((link) => link.classList.toggle("is-active", link.getAttribute("href") === `#${visible.target.id}`));
473  }, { rootMargin: "-18% 0px -70%", threshold: [0, .2, .5] });
474  sections.forEach((section) => observer.observe(section));
475}

477async function init() {
478  bindInteractions();
479  try {
480    const response = await fetch(CATALOG_URL, { cache: "no-store" });
481    if (!response.ok) throw new Error(`catalog fetch failed: ${response.status}`);
482    state.catalog = await response.json();
483    renderOverview(state.catalog);
484    renderEcosystem();
485    renderOfficialResources(state.catalog);
486    renderInfrastructure(state.catalog);
487    renderRevenue(state.catalog);
488    renderBuildTracks(state.catalog);
489    renderResearch(state.catalog);
490    renderGates(state.catalog);
491  } catch (error) {
492    renderError(error);
493  }
494}
